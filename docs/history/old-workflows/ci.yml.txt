name: CI

on:
  pull_request:
    branches: [approved]
  push:
    branches: [approved]

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  js-syntax:
    name: JavaScript syntax
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
      - name: node --check app.js / drawing.js / geometry.js
        run: |
          node --check "Project-Check-MVP-Master-Build/app.js"
          node --check "Project-Check-MVP-Master-Build/drawing.js"
          node --check "Project-Check-MVP-Master-Build/geometry.js"

  html-validity:
    name: HTML validity
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          # html-validate@11.8.0 requires Node ^22.22 || >=24.8; Node 20 crashed
          # before it ever parsed the file (fs.globSync is not a function).
          node-version: "24"
      - name: Write html-validate config (recommended preset, no repo file)
        run: echo '{"extends":["html-validate:recommended"]}' > "$RUNNER_TEMP/htmlvalidate.json"
      - name: html-validate index.html
        run: npx --yes html-validate@11.8.0 --config "$RUNNER_TEMP/htmlvalidate.json" "Project-Check-MVP-Master-Build/index.html"

  css-validity:
    name: CSS validity
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
      - name: Write stylelint config (syntax check only, no style rules, no repo file)
        run: echo '{"rules":{}}' > "$RUNNER_TEMP/stylelintrc.json"
      - name: stylelint styles.css (syntax check only, no style rules)
        run: |
          npm install --no-save stylelint@17.14.1
          npx stylelint --config "$RUNNER_TEMP/stylelintrc.json" "Project-Check-MVP-Master-Build/styles.css"

  logo-asset:
    name: Logo asset present
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Confirm approved logo asset exists
        run: |
          FILE="Project-Check-MVP-Master-Build/assets/project-check-logo.png"
          if [ ! -s "$FILE" ]; then
            echo "Missing or empty: $FILE" >&2
            exit 1
          fi
          echo "Found: $FILE"
