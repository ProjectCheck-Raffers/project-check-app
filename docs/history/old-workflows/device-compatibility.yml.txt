name: Device compatibility & security regression suite

on:
  pull_request:
    branches: [approved]
    paths:
      - "Project-Check-MVP-Master-Build/**"
      - "tests/**"
      - ".github/workflows/**"
  push:
    branches: [approved]
    paths:
      - "Project-Check-MVP-Master-Build/**"
      - "tests/**"
      - ".github/workflows/**"
  workflow_dispatch:
    inputs:
      suite:
        description: Test scope
        required: true
        default: profile
        type: choice
        options:
          - profile
          - full

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  playwright:
    if: ${{ github.event_name == 'pull_request' || github.event_name == 'push' || (github.event_name == 'workflow_dispatch' && inputs.suite == 'full') }}
    name: Playwright (responsive, journey, security, a11y, error-handling, live-map)
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "22"
      - name: Isolate browser tests from live Supabase
        run: echo 'window.PROJECT_CHECK_PLATFORM_CONFIG = {};' > Project-Check-MVP-Master-Build/platform-config.js
      - name: Install test dependencies
        working-directory: tests/playwright
        run: npm ci
      - name: Install Playwright browsers (Chromium, WebKit, Firefox)
        working-directory: tests/playwright
        run: npx playwright install --with-deps chromium webkit firefox
      - name: Check Firefox WebGL renderer
        working-directory: tests/playwright
        run: xvfb-run -a node verify-firefox-webgl.js
      - name: Run full Playwright suite
        working-directory: tests/playwright
        # Provide a virtual display for the Firefox WebGL rendering candidate.
        # xvfb is included by the pinned Playwright --with-deps installation.
        run: xvfb-run -a npx playwright test
      - name: Upload Playwright HTML report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          # Include raw traces when a failed/interrupted run has not produced
          # its final HTML report. Upload still requires an available runner.
          path: |
            /tmp/project-check-playwright-html-report/
            /tmp/project-check-playwright-test-results/
          retention-days: 14

  profile-targeted:
    if: ${{ github.event_name != 'workflow_dispatch' || inputs.suite == 'profile' || inputs.suite == 'full' }}
    name: Profile navigation (desktop and mobile)
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "22"
      - name: Isolate browser tests from live Supabase
        run: echo 'window.PROJECT_CHECK_PLATFORM_CONFIG = {};' > Project-Check-MVP-Master-Build/platform-config.js
      - name: Install committed test dependencies
        working-directory: tests/playwright
        run: npm ci
      - name: Install Chromium
        working-directory: tests/playwright
        run: npx playwright install --with-deps chromium
      - name: Run two targeted Profile tests
        working-directory: tests/playwright
        env:
          PROFILE_TARGETED_SPEC: |
            const { test, expect } = require('@playwright/test');
            async function verifyProfileJourney(page, viewport, label) {
              await page.setViewportSize(viewport);
              const account = {id:'profile-test', name:'Profile Customer', company:'Profile Company', companyAddress:'1 Test Road, WA12 9XX', email:`profile-${label}@example.com`, mobile:'07700 900001'};
              await page.addInitScript(account => {
                localStorage.setItem('projectcheck_master_v1', JSON.stringify({user:{id:account.id,email:account.email}, users:[account], projects:[], stage3ImpactRegister:[]}));
              }, account);
              await page.goto('/#dashboard');
              await expect(page.getByRole('heading', {name:'My Projects'})).toBeVisible();
              await page.locator('#menuBtn').click();
              await page.locator('#siteDrawer').getByRole('button', {name:'Profile & Settings', exact:true}).click();
              await expect(page).toHaveURL(/#profile$/);
              await expect(page.getByRole('heading', {name:'Profile overview',exact:true})).toBeVisible();
              for (const value of [account.name,account.company,account.companyAddress,account.email,account.mobile]) {
                await expect(page.locator('.profile-details dd').filter({hasText:value})).toBeVisible();
              }
              await expect(page.getByLabel('Project name',{exact:true})).toHaveCount(0);
              await page.getByRole('button',{name:'Edit profile',exact:true}).click();
              await expect(page.getByRole('heading',{name:'Edit profile',exact:true})).toBeVisible();
              await expect(page.getByLabel('Full name',{exact:true})).toHaveValue(account.name);
              await page.getByRole('button',{name:'Cancel',exact:true}).click();
              await expect(page.getByRole('heading',{name:'Profile overview',exact:true})).toBeVisible();
              await page.locator('.customer-heading').getByRole('button',{name:'My Projects',exact:true}).click();
              await expect(page).toHaveURL(/#dashboard$/);
            }

            test('Profile opens from My Projects on desktop', async ({ page }) => {
              await verifyProfileJourney(page, { width: 1280, height: 800 }, 'desktop');
            });

            test('Profile opens from My Projects on mobile', async ({ page }) => {
              await verifyProfileJourney(page, { width: 390, height: 844 }, 'mobile');
            });
        run: |
          node -e "require('node:fs').writeFileSync('profile-transition-navigation.spec.js', process.env.PROFILE_TARGETED_SPEC)"
          npx playwright test profile-transition-navigation.spec.js --project=functional-chromium --workers=1
      - name: Upload targeted Profile report
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: profile-targeted-report
          path: /tmp/project-check-playwright-html-report/
          retention-days: 14
