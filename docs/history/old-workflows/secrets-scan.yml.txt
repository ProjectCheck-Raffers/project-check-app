name: Secrets scan

on:
  pull_request:
    branches: [approved]
  push:
    branches: [approved]

permissions:
  contents: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  gitleaks:
    name: gitleaks (pinned CLI, full history)
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Install gitleaks 8.30.1 (pinned)
        # -f makes curl exit non-zero on an HTTP error response (e.g. a 404
        # if the pinned version/URL ever goes stale) instead of silently
        # writing the error page to gitleaks.tar.gz and letting tar fail
        # with a confusing "not in gzip format" instead of the real cause.
        run: |
          curl -fsSL -o gitleaks.tar.gz https://github.com/gitleaks/gitleaks/releases/download/v8.30.1/gitleaks_8.30.1_linux_x64.tar.gz
          tar -xzf gitleaks.tar.gz gitleaks
          ./gitleaks version
      - name: Scan full git history for secrets
        run: ./gitleaks detect --source . --no-banner --redact
